然后配置prometheus资源，检查了下：每个资源都在
```yaml
# npu-exporter-service.yaml
apiVersion: v1
kind: Service
metadata:
  name: npu-exporter
  namespace: monitoring
  labels:
    app: npu-exporter
spec:
  type: ClusterIP
  ports:
  - name: metrics
    port: 8082
    targetPort: 8082
---
apiVersion: v1
kind: Endpoints
metadata:
  name: npu-exporter
  namespace: monitoring
subsets:
- addresses:
  - ip: 172.22.0.104  # 替换为节点IP
  ports:
  - name: metrics
    port: 8082
---
# npu-exporter-servicemonitor.yaml
apiVersion: monitoring.coreos.com/v1
kind: ServiceMonitor
metadata:
  name: npu-exporter
  namespace: monitoring
spec:
  endpoints:
  - port: metrics  # 对应Service端口名称
    interval: 10s
  namespaceSelector:
    matchNames:
    - monitoring
  selector:
    matchLabels:
      app: npu-exporter  # 需与Service的Label匹配
```
